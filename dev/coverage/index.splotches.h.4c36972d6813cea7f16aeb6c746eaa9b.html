21 /// Swarms of splotches thrown by new craters, the low-velocity secondaries of their ejecta
22 /// (Robinson et al. 2015; Speyerer et al. 2016; Housen and Holsapple 2011), off by default.
23 struct SplotchSwarms {
24 bool enabled = false; ///< Whether new craters throw swarms.
25 double craterRate = 4.84e-6; ///< New craters of at least 10 m, per km^2 per year.
26 double craterSlope = 3.06; ///< Cumulative slope of the new craters; positive.
27 double minCraterDiameter = 10.0; ///< Smallest crater with a swarm (m); positive.
28 double maxCraterDiameter = 50.0; ///< Largest (m); above the smallest.
29 double referenceDiameter = 18.8; ///< Diameter of the reference crater (m).
30 double referenceCount = 248.0; ///< Splotches of the reference crater; positive.
31 double mu = 0.41; ///< Velocity exponent of the target, in (0, 1).
32 double minRange = 4300.0; ///< Nearest landing distance (m); positive.
33 double maxRange = 30000.0; ///< Farthest landing distance (m); above the nearest.
34 /// Diameter of a swarm splotch (m); positive and bounded.
35 ValueDistribution splotchDiameter{ValueDistribution::LogNormal(6.0, 0.6).Clamped(2.0, 22.0)};
36 double darkShare = 239.0 / 248.0; ///< Share darker than the ground, in [0, 1].
37 double brightShare = 3.0 / 248.0; ///< Share brighter; the rest are mixed.
38 /// Reflectance change of a dark swarm splotch; above -1 and bounded.
39 ValueDistribution darkChange{ValueDistribution::Constant(-0.04)};
40 /// Reflectance change of a bright swarm splotch; above -1 and bounded.
41 ValueDistribution brightChange{ValueDistribution::Uniform(0.04, 0.12)};
42 double wedgeShare = 0.2; ///< Share of wedges pointing back to the crater, [0, 1].
43 double wedgeElongation = 2.0; ///< Long over short axis of a wedge; at least 1.
44 /// New craters given by hand, such as an observed impact, each throwing a swarm like the
45 /// generated ones; IDs of 0 come from `ManualCraterId`.

71 /// Splotches after Speyerer et al. (2016): formed at `rateCoefficient` d^-`rateExponent` per km^2
72 /// per year (d in km), each lasting `lifetimeMyr`; dark, bright or mixed.
73 struct SplotchAlbedoOptions {
74 std::uint32_t seed = 0; ///< Seed of where splotches fall and what they are.
75 double rateCoefficient = 1.483e-11; ///< Formation rate coefficient, per km^2 per year.
76 double rateExponent = 4.144; ///< Cumulative slope of the formation rate; positive.
77 double lifetimeMyr = 0.081; ///< How long a splotch lasts, fading linearly (Myr).
78 double minDiameter = 1.0; ///< Smallest splotch (m); the fit is extrapolated below 10.
79 double maxDiameter = 40.0; ///< Largest splotch (m), above `minDiameter`.
80 double darkShare = 0.907; ///< Share of dark splotches, in [0, 1].
81 double brightShare = 0.074; ///< Share of bright ones; the rest are mixed.
82 /// Reflectance change of a dark splotch, relative to the ground before it; above -1.
83 ValueDistribution darkChange{ValueDistribution::Constant(-0.04)};
84 /// Reflectance change of a bright splotch; above -1.
85 ValueDistribution brightChange{ValueDistribution::Constant(0.10)};
86 /// Share of splotches on slopes steeper than `steepSlopeDegrees` that are bright, in [0, 1];
87 /// unset keeps `brightShare`. Bright splotches are found mostly on such slopes, where mature
88 /// regolith is thin (Speyerer et al. 2016), but the sources give no share; needs the ground.
89 std::optional<double> steepBrightShare;
90 double steepSlopeDegrees = 15.0; ///< Slope from which ground is steep, in (0, 90) degrees.
91 double softness = 0.3; ///< Share of the radius over which the edge fades, in (0, 1].
92 double raggedness = 0.15; ///< Share of the radius the edge's lobes pull in, in [0, 1).
93 ResolutionFade fade{}; ///< How splotches too small to resolve fade out.
94 SplotchSwarms swarms{}; ///< Swarms around new craters; off by default.
95
96 /// The cumulative number of splotches of diameter at least 1 m per m^2 at any time.
97 double Coefficient() const;
98
99 bool operator==(const SplotchAlbedoOptions& other) const {
100 return seed == other.seed && rateCoefficient == other.rateCoefficient &&
101 rateExponent == other.rateExponent && lifetimeMyr == other.lifetimeMyr &&
102 minDiameter == other.minDiameter && maxDiameter == other.maxDiameter &&
103 darkShare == other.darkShare && brightShare == other.brightShare &&
104 darkChange == other.darkChange && brightChange == other.brightChange &&
105 steepBrightShare == other.steepBrightShare &&
106 steepSlopeDegrees == other.steepSlopeDegrees && softness == other.softness &&
107 raggedness == other.raggedness && fade == other.fade && swarms == other.swarms;
108 }
109 bool operator!=(const SplotchAlbedoOptions& other) const {
110 return !(*this == other);
111 }
