17 /// Self-affine roughness: the RMS height difference at a lag L is nu(L) = `deviation` (L /
18 /// `cellSize`)^H, so the RMS slope over a baseline b is nu(b) / b = s(c) (b / c)^(H - 1)
19 /// (Shepard et al. 2001; Rosenburg et al. 2011, eq. 4, https://doi.org/10.1029/2010JE003716).
20 1 struct BaseRoughness {
21 double hurst = 1.0; ///< Hurst exponent H, in (0, 1].
22 double deviation = 0.0; ///< RMS height difference at a lag of `cellSize`, in meters.
23 double cellSize = 1.0; ///< Lag `deviation` is given at, in meters.
24
25 /// The RMS height difference in meters at a lag of `lag` meters.
26 double Deviation(double lag) const;
27
28 /// The RMS slope (rise over run) over a baseline of `baseline` meters.
29 double RmsSlope(double baseline) const;
30 };
