16 /// How rocks, brighter than mature regolith (Holt 1981; Clark et al. 2002; Marshal et al. 2023),
17 /// brighten the albedo: resolved boulders where they stand, unresolved rock by a per-sample draw
18 /// of the share it covers.
19 5 struct RockAlbedoOptions {
20 std::uint32_t seed = 0; ///< Seed of the per-sample draws of unresolved rock.
21 double reflectanceRatio = 2.5; ///< Rock over regolith reflectance; positive.
22 bool unresolved = true; ///< Whether rocks too small to resolve brighten samples.
23
24 bool operator==(const RockAlbedoOptions& other) const {
25 return seed == other.seed && reflectanceRatio == other.reflectanceRatio &&
26 unresolved == other.unresolved;
27 }
28 bool operator!=(const RockAlbedoOptions& other) const {
29 return !(*this == other);
30 }
31 };
